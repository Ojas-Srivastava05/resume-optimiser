\section{How HireVue works here}
\begin{itemize}
  \item Browser-based recorded interview (no live human)
  \item Short prep time, then you record video/audio answer
  \item \textbf{3 attempts} per question --- use attempt 1 as rehearsal if needed
  \item AI/human reviewers score clarity, structure, confidence
\end{itemize}

\section{Setup checklist}
\begin{itemize}
  \item Plain background, face well-lit, camera at eye level
  \item Quiet room; phone on silent
  \item Business casual (collared shirt)
  \item Bullet notes on sticky note \textbf{off camera} --- don't read verbatim
  \item Speak 90--120 seconds max per answer unless prompt says shorter
\end{itemize}

\section{Practice drill}
\begin{enumerate}
  \item Record each answer once --- watch without sound (body language check)
  \item Record again --- aim for \textbf{no filler} (um, basically, like)
  \item Third pass --- time each answer; cut if over 2 min
\end{enumerate}

\section{Avoid}
\begin{itemize}
  \item Memorized robotic delivery --- use bullets, not script
  \item Badmouthing college/teammates
  \item ``I just need any internship'' --- show GE-specific motivation
  \item Reading from screen --- eyes darting kills HireVue scores
\end{itemize}
