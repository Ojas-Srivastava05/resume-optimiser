\section{3) OA --- CS MCQ flash (GFG)}
\begin{itemize}
  \item \textbf{OOP:} 4 pillars; overloading vs overriding; virtual functions; copy ctor
  \item \textbf{OS:} process vs thread; deadlock (4); mutex vs semaphore; paging
  \item \textbf{DBMS:} ACID; 1NF--3NF; PK/FK; INNER vs LEFT JOIN; index = faster read, slower write
  \item \textbf{CN:} OSI layers; TCP vs UDP; 3-way handshake; DHCP, DNS
  \item \textbf{DSA theory:} stack LIFO, queue FIFO; BST property; BFS vs DFS use-cases
\end{itemize}

\section{4) Kadane + Two Sum (dry-run) + OOP wrap}
\begin{itemize}
  \item \textbf{Kadane:} \texttt{[-2,1,-3,4,-1,2,1,-5,4]} $\rightarrow$ max = \textbf{6}, subarray \texttt{[4,-1,2,1]}
  \item \textbf{Two Sum:} one pass hashmap of \texttt{target - x}; O(n) time O(n) space
  \item \textbf{EID OOP:} class \texttt{TwoSumSolver\{solve(nums,target)\}}; class \texttt{TreeNode} + \texttt{leafSum(root)}
  \item \textbf{First unique char:} freq map, two passes; or single pass with ordered dict
\end{itemize}

\section{5) OOP --- speak + code}
\begin{itemize}
  \item \textbf{Encapsulation} hide data (private fields + public methods)
  \item \textbf{Inheritance} reuse (Django models / class hierarchy)
  \item \textbf{Polymorphism} same interface, different behavior (runtime via virtual/override)
  \item \textbf{Compile-time} overload; \textbf{runtime} override --- EEDP FTE asked \textbf{implement live}
  \item Tie to \textbf{Beneath The Blue:} User model, auth views, quiz classes
\end{itemize}

\section{6) OS / DBMS / CN one-liners}
\begin{itemize}
  \item \textbf{Race condition:} two threads read/write shared data --- fix with mutex/lock
  \item \textbf{Normalization:} reduce redundancy; 3NF removes transitive dependency
  \item \textbf{SQL:} \texttt{SELECT * FROM t ORDER BY sal DESC LIMIT 1 OFFSET 2} (3rd highest trick)
  \item \textbf{3-tier:} presentation (React/HTML) $\rightarrow$ business logic (Django) $\rightarrow$ data (MySQL)
  \item \textbf{REST:} JSON over HTTP; GET read, POST create, PATCH update
\end{itemize}
